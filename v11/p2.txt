    const AC=window.AudioContext||window.webkitAudioContext;

function unlockAudio(){
  const ac=initAudio();if(!ac)return;
  try{
    const b=ac.createBuffer(1,1,22050),src=ac.createBufferSource(),g=ac.createGain();
    src.buffer=b;g.gain.value=.00001;src.connect(g);g.connect(ac.destination);src.start(0);
  }catch(e){}
  const mark=()=>{audioReady=ac.state==='running';if(audioReady&&!audioUnlocked){audioUnlocked=true;tone(520,.045,'sine',.018,120)}};
  try{
    if(ac.state==='suspended'){const r=ac.resume();if(r&&r.then)r.then(mark).catch(()=>{})}else mark();
    ac.onstatechange=mark;
  }catch(e){}
}

}
